export interface BlogPost {
  id: string;
  title: string;
  slug: string;
  excerpt: string;
  content: string;
  htmlContent?: string;
  imageUrl: string;
  thumbnailUrl?: string;
  publishedAt: string;
  updatedAt: string;
  readTime: number;
  category: BlogCategory;
  tags: BlogTag[];
  status: 'draft' | 'published' | 'archived';
  featured: boolean;
  viewCount: number;
  likeCount: number;
  commentCount: number;
  seoMetadata: SeoMetadata;
  relatedPosts?: string[];
  tableOfContents?: TableOfContentsItem[];
  socialSharing: SocialSharingConfig;
}
